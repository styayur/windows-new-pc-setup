# 语言环境
- type: Microsoft.WinGet/Package
  name: OpenJDK21
  properties:
    id: Microsoft.OpenJDK.21
    source: winget
    useLatest: true

- type: Microsoft.DSC.Transitional/RunCommandOnSet
  name: JavaHome
  dependsOn:
  - OpenJDK21
  properties:
    executable: C:\Windows\System32\WindowsPowerShell\v1.0\powershell.exe
    arguments:
      "0": -NoProfile
      "1": -ExecutionPolicy
      "2": Bypass
      "3": -Command
      "4": |
        $java = Get-Command java.exe -ErrorAction SilentlyContinue
        if (-not $java) { throw 'java.exe not found' }
        $jdkHome = Split-Path (Split-Path $java.Source -Parent) -Parent
        [Environment]::SetEnvironmentVariable('JAVA_HOME', $jdkHome, 'Machine')
        $path = [Environment]::GetEnvironmentVariable('Path', 'Machine')
        if ($path -notlike "*$jdkHome\bin*") {
          [Environment]::SetEnvironmentVariable('Path', "$path;$jdkHome\bin", 'Machine')
        }
      treatAsArray: true
  metadata:
    winget:
      securityContext: elevated
    description: 配置 JAVA_HOME
