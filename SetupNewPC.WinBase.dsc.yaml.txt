# 3. Virtualization and WSL prerequisite gate
# ---------------------------------------------------------------------------
- type: Microsoft.DSC.Transitional/RunCommandOnSet
  name: EnableWslPrerequisites
  dependsOn:
  - ProcessExplorer
  properties:
    executable: C:\Windows\System32\WindowsPowerShell\v1.0\powershell.exe
    arguments:
      "0": -NoProfile
      "1": -ExecutionPolicy
      "2": Bypass
      "3": -Command
      "4": |
        $required = @('VirtualMachinePlatform', 'Microsoft-Windows-Subsystem-Linux')
        foreach ($name in $required) {
          $feature = Get-WindowsOptionalFeature -Online -FeatureName $name -ErrorAction Stop
          if ($feature.State -notin @('Enabled', 'EnablePending')) {
            Enable-WindowsOptionalFeature -Online -FeatureName $name -All -NoRestart -ErrorAction Stop | Out-Null
          }
        }

        $cpu = Get-CimInstance Win32_Processor | Select-Object -First 1
        $computer = Get-CimInstance Win32_ComputerSystem
        $hardwareBlocked = $computer.HypervisorPresent -eq $false -and
          $cpu.VirtualizationFirmwareEnabled -eq $false -and
          $cpu.VMMonitorModeExtensions -eq $false
        if ($hardwareBlocked) {
          Write-Warning 'CPU virtualization is disabled in firmware. Enable VT-x/AMD-V in BIOS/UEFI.'
        }

        $pending = @()
        if (Test-Path 'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending') {
          $pending += 'Component Based Servicing'
        }
        if (Test-Path 'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired') {
          $pending += 'Windows Update'
        }
        $sm = Get-ItemProperty 'HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager' -ErrorAction SilentlyContinue
        if ($sm.PendingFileRenameOperations) { $pending += 'Pending file rename' }

        if ($pending.Count -gt 0) {
          Write-Host "REBOOT REQUIRED: $($pending -join ', ')"
          Write-Host 'After reboot, apply SetupNewPC.Development.dsc.yaml.'
        } else {
          Write-Host 'WSL/VMP prerequisites are active; no reboot is currently required.'
        }
      treatAsArray: true
  metadata:
    winget:
      securityContext: elevated
    description: Enable VirtualMachinePlatform and Microsoft-Windows-Subsystem-Linux without automatic reboot.
